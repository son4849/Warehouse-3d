    <script>
        const nvlDatabase = [
            { code: "PCB-MAIN-A1", name: "Bo mạch chính A1", room: 1, rack: "G1", tier: 3, desc: "Khay tĩnh điện - Hàng cao cấp" },
            { code: "FPCB-FL-02", name: "Cáp dẻo màn hình", room: 1, rack: "G2", tier: 2, desc: "Thùng 500 pcs" },
            { code: "IC-PWR-5V", name: "IC Nguồn 5V", room: 1, rack: "G3", tier: 1, desc: "Cuộn SMD" },
            { code: "RES-0603-10K", name: "Trở dán 10k Ohm", room: 2, rack: "G4", tier: 4, desc: "Giá B linh kiện nhỏ" },
            { code: "CONN-TYPE-C", name: "Cổng sạc Type-C", room: 2, rack: "G5", tier: 2, desc: "Thùng carton 1000 con" }
        ];

        const roomsConfig = {
            1: [
                { id: "G1", label: "GIÁ A1" },
                { id: "G2", label: "GIÁ A2" },
                { id: "G3", label: "GIÁ A3" }
            ],
            2: [
                { id: "G4", label: "GIÁ B1" },
                { id: "G5", label: "GIÁ B2" },
                { id: "G6", label: "GIÁ B3" }
            ]
        };

        let currentRoom = 1;

        function renderRoom3D(roomId) {
            currentRoom = roomId;
            const canvas = document.getElementById("roomCanvas");
            canvas.innerHTML = "";

            document.getElementById("btnRoom1").classList.toggle("active", roomId === 1);
            document.getElementById("btnRoom2").classList.toggle("active", roomId === 2);

            const racks = roomsConfig[roomId] || [];
            racks.forEach(r => {
                const rackDiv = document.createElement("div");
                rackDiv.className = "rack-3d";
                rackDiv.id = rack3d-${r.id};

                rackDiv.innerHTML = `
                    <div class="rack-label-3d">${r.label}</div>
                    <div class="rack-tier" id="tier-${r.id}-4">Tầng 4</div>
                    <div class="rack-tier" id="tier-${r.id}-3">Tầng 3</div>
                    <div class="rack-tier" id="tier-${r.id}-2">Tầng 2</div>
                    <div class="rack-tier" id="tier-${r.id}-1">Tầng 1</div>
                `;
                canvas.appendChild(rackDiv);
            });
        }

        function renderItemList(items) {
            const listEl = document.getElementById("itemsList");
            listEl.innerHTML = "";
            items.forEach(item => {
                const div = document.createElement("div");
                div.className = "item-row";
                div.onclick = () => selectItem(item);
                div.innerHTML = `
                    <div class="item-code">${item.code}</div>
                    <div class="item-desc">${item.name} | <b>Phòng ${item.room} - Giá ${item.rack} (Tầng ${item.tier})</b></div>
                `;
                listEl.appendChild(div);
            });
        }

        function selectItem(item) {
            if (currentRoom !== item.room) {
                renderRoom3D(item.room);
            }

            document.querySelectorAll(".rack-3d").forEach(el => el.classList.remove("highlight-3d"));
            document.querySelectorAll(".rack-tier").forEach(el => el.classList.remove("highlight-tier"));

            setTimeout(() => {
                const rackEl = document.getElementById(rack3d-${item.rack});
                const tierEl = document.getElementById(tier-${item.rack}-${item.tier});

                if (rackEl && tierEl) {
                    rackEl.classList.add("highlight-3d");
                    tierEl.classList.add("highlight-tier");
                }
            }, 100);

            document.getElementById("infoContent").innerHTML = `
                <div class="info-code">${item.code}</div>
                <div><b>Tên NVL:</b> ${item.name}</div>
                <div style="margin-top:6px;">
                    <span class="badge badge-room">Phòng Kho ${item.room}</span>
                    <span class="badge badge-rack">Giá ${item.rack} - Tầng ${item.tier}</span>
                </div>
                <div style="font-size:0.8rem; color:#94a3b8; margin-top:6px;">${item.desc}</div>
            `;
        }

        function switchRoom(roomId) {
            renderRoom3D(roomId);
        }

        function handleSearch() {
            const query = document.getElementById("searchInput").value.trim().toLowerCase();
            const filtered = nvlDatabase.filter(i => 
                i.code.toLowerCase().includes(query) || 
                i.name.toLowerCase().includes(query) ||
                i.rack.toLowerCase().includes(query)
            );
            renderItemList(filtered);
            if (filtered.length === 1) selectItem(filtered[0]);
        }

        renderRoom3D(1);
        renderItemList(nvlDatabase);
    </script>
